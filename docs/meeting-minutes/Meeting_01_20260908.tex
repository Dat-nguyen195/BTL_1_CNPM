\section{Mục tiêu \& Yêu cầu cần đạt được sau Sprint 1 \& 2 (13/09 -- 20/09)}

Giai đoạn đầu tập trung hoàn thành toàn bộ nền tảng phân tích yêu cầu phục vụ Submission \#1. Các kết quả bắt buộc cần đạt:

\subsection{Understanding Project's Context (Hiểu rõ bối cảnh dự án)}

\begin{itemize}[leftmargin=2em, itemsep=2pt]
  \item Thấu hiểu sâu sắc bài toán điều phối phương tiện điện (Smart E-Mobility Hub) tại Khu đô thị ĐHQG-HCM.
  \item Xác định rõ bối cảnh kết nối \textbf{Ga Metro số 1}, \textbf{KTX Khu A/B}, phạm vi quản lý trạm sạc và bãi đỗ phương tiện.
  \item Phân tích toàn diện kỳ vọng của các bên liên quan (\textit{Stakeholders}):
    \begin{itemize}[leftmargin=1.5em, itemsep=1pt]
      \item \textbf{Sinh viên:} Đặt xe dùng chung, nhận/trả xe linh hoạt, gửi xe cá nhân, tra cứu lịch sạc.
      \item \textbf{Operator:} Giám sát trạng thái hub, điều phối phương tiện cân bằng tải, ghi nhận sự cố.
      \item \textbf{Nhà trường / Ban Quản lý:} Phân tích nhu cầu, mô phỏng kịch bản What-if, tối ưu hóa hạ tầng.
    \end{itemize}
\end{itemize}

\subsection{Functional \& Non-functional Requirements (Hệ thống Yêu cầu)}

\textbf{Functional Requirements:}
\begin{itemize}[leftmargin=2em, itemsep=2pt]
  \item Xác định danh mục chức năng tổng thể và ranh giới nghiệp vụ của 7 phân hệ (SS1 đến SS7).
  \item Mỗi thành viên đặc tả chi tiết ít nhất 02 Use-case cá nhân trong phân hệ được phân công.
\end{itemize}

\textbf{Non-functional Requirements:}
\begin{itemize}[leftmargin=2em, itemsep=2pt]
  \item \textbf{Performance:} Thời gian phản hồi $\leq 2.0$ giây cho các tác vụ đặt xe, điều phối và cập nhật trạng thái.
  \item \textbf{Reliability:} Đảm bảo tính nhất quán dữ liệu in-memory (\texttt{st.session\_state}), tránh race-condition khi nhiều người cùng đặt 1 xe.
  \item \textbf{Availability:} Prototype Streamlit hoạt động ổn định 24/7 trong môi trường dev.
  \item \textbf{Usability:} Giao diện tối ưu hóa cho màn hình di động (Sinh viên) và màn hình Dashboard rộng (Operator).
\end{itemize}

\subsection{Use-case Diagram (Sơ đồ Use-case toàn hệ thống \& Phân hệ)}

\begin{itemize}[leftmargin=2em, itemsep=2pt]
  \item \textbf{System-wide Use-case Diagram:} Nguyễn Thành Đạt \& Phạm Tấn Đạt chủ trì xây dựng sơ đồ tổng thể bao phủ toàn bộ 7 phân hệ -- 4 Actor chính: \textit{Sinh viên, Operator, Kỹ thuật viên, Hệ thống mô phỏng IoT}; quan hệ \texttt{<<include>>}, \texttt{<<extend>>}. Lưu tại \texttt{docs/diagrams/system/}.
  \item \textbf{Cấu trúc nộp:} File nguồn \texttt{.drawio}/\texttt{.puml} + ảnh \texttt{.png} ($\geq 300$ DPI), đặt tên theo cú pháp \texttt{<MSSV>\_<Mã\_Use\_Case>.[drawio|png]}.
\end{itemize}

\subsection{Đặc tả kịch bản Use-case chi tiết cá nhân (Use-case detail/scenario -- Individual Work)}

\begin{itemize}[leftmargin=2em, itemsep=2pt]
  \item \textbf{Tính chất bắt buộc:} Hạng mục đánh giá \textbf{đóng góp cá nhân bắt buộc (Individual Work)} theo quy định đề bài BTL. Mỗi thành viên phải tự mình phân tích và đặc tả chi tiết \textbf{02 Use-cases} thuộc phân hệ được phân công.
  \item \textbf{Tiêu chuẩn cấu trúc bảng (Template 14 trường theo \texttt{mau.png}):}
    \begin{enumerate}[label=\alph*., leftmargin=1.5em, itemsep=0pt, parsep=0pt]
      \item \textit{Use Case ID \& Use Case Name} (đúng chuẩn định danh \texttt{UC\_SS<X>\_<YY>}).
      \item \textit{Created By / Last Updated By} \& \textit{Date Created / Date Last Updated} (cột đôi hành chính).
      \item \textit{Actors:} Phân định rõ Primary Actor và Secondary Actors (Hệ thống IoT, Session State).
      \item \textit{Description:} Tóm tắt ngắn gọn mục tiêu nghiệp vụ trong 2--3 câu.
      \item \textit{Trigger:} Sự kiện hoặc hành động trực tiếp kích hoạt Use-case.
      \item \textit{Preconditions \& Postconditions:} Điều kiện tiên quyết và trạng thái sau khi hoàn tất (nêu rõ trạng thái phương tiện: \texttt{AVAILABLE}, \texttt{RESERVED}, \texttt{IN\_USE}, \texttt{CHARGING}, \texttt{dispatched}, \texttt{incident}).
      \item \textit{Normal Flow:} Trình tự các bước thực thi thành công (5--7 bước rõ ràng giữa Người dùng $\leftrightarrow$ Hệ thống).
      \item \textit{Alternative Flows:} Tối thiểu 1--2 luồng rẽ nhánh thực tế (lọc pin, đổi xe, đổi hub...).
      \item \textit{Exceptions:} Tối thiểu 1--2 luồng ngoại lệ và xử lý sự cố (hết hạn giữ chỗ, hub đầy chỗ, lỗi kết nối...).
      \item \textit{Notes and Issues:} Ràng buộc phi chức năng đặc thù, thời gian xử lý, tính nhất quán dữ liệu.
    \end{enumerate}
  \item \textbf{Quy chuẩn lưu trữ file:} Mỗi cá nhân lưu trữ file tại thư mục tương ứng:\newline
  \texttt{docs/diagrams/Use-case detail-scenario/SS<X>/UC\_SS<X>\_<HoVaTen>.[md|docx]} \newline
  \textit{(Khuyến nghị có cả bản \texttt{.md} hiển thị trực tiếp trên GitHub và bản \texttt{.docx} phục vụ ghép tài liệu nộp BKeL)}.
\end{itemize}
